# ============================================================
# Улётная Парковка — оптимизация .htaccess
# ============================================================
# ВСТАВИТЬ В НАЧАЛО /www/uletnayaparkovka.ru/.htaccess
# (перед блоком # BEGIN WordPress)
#
# Что делает:
#   - Brotli + gzip компрессия
#   - Жёсткий кеш статики (1 год + immutable)
#   - WebP fallback (отдаёт .webp если есть и браузер поддерживает)
#   - Безопасностные заголовки
#   - Запрещает доступ к чувствительным файлам WP
# ============================================================

# ----- 1. Compression -----
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/plain text/xml text/css text/javascript
  AddOutputFilterByType DEFLATE application/javascript application/x-javascript application/json
  AddOutputFilterByType DEFLATE application/xml application/rss+xml application/atom+xml
  AddOutputFilterByType DEFLATE image/svg+xml font/ttf font/otf font/woff font/woff2
</IfModule>

<IfModule mod_brotli.c>
  AddOutputFilterByType BROTLI_COMPRESS text/html text/css text/javascript application/javascript application/json image/svg+xml
</IfModule>

# ----- 2. Aggressive cache for static -----
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresDefault                              "access plus 1 month"
  ExpiresByType text/html                     "access plus 5 minutes"
  ExpiresByType text/css                      "access plus 1 year"
  ExpiresByType application/javascript        "access plus 1 year"
  ExpiresByType application/x-javascript      "access plus 1 year"
  ExpiresByType image/jpeg                    "access plus 1 year"
  ExpiresByType image/png                     "access plus 1 year"
  ExpiresByType image/webp                    "access plus 1 year"
  ExpiresByType image/svg+xml                 "access plus 1 year"
  ExpiresByType image/x-icon                  "access plus 1 year"
  ExpiresByType font/woff2                    "access plus 1 year"
  ExpiresByType font/woff                     "access plus 1 year"
  ExpiresByType font/ttf                      "access plus 1 year"
  ExpiresByType application/font-woff         "access plus 1 year"
  ExpiresByType application/font-woff2        "access plus 1 year"
</IfModule>

<IfModule mod_headers.c>
  # immutable для всех файлов с хешем/версией
  <FilesMatch "\.(css|js|webp|jpe?g|png|gif|svg|woff2?|ttf|otf|ico)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
    Header unset Set-Cookie
    Header unset Pragma
  </FilesMatch>
</IfModule>

# ----- 3. WebP rewrite — отдаём .webp если есть и браузер поддерживает -----
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteCond %{HTTP_ACCEPT} image/webp
  RewriteCond %{REQUEST_FILENAME} \.(jpe?g|png)$
  RewriteCond %{REQUEST_FILENAME}.webp -f
  RewriteRule ^(.+)\.(jpe?g|png)$ $1.$2.webp [T=image/webp,E=accept:1,L]
</IfModule>

<IfModule mod_headers.c>
  <FilesMatch "\.(jpe?g|png)$">
    Header append Vary Accept env=REDIRECT_accept
  </FilesMatch>
</IfModule>

# ----- 4. Security headers -----
<IfModule mod_headers.c>
  Header set X-Frame-Options "SAMEORIGIN"
  Header set X-Content-Type-Options "nosniff"
  Header set Referrer-Policy "strict-origin-when-cross-origin"
  Header set Permissions-Policy "geolocation=(self), camera=(), microphone=()"
</IfModule>

# ----- 5. Block sensitive files -----
<FilesMatch "(^\.|wp-config\.php|readme\.html|license\.txt|xmlrpc\.php)">
  Order allow,deny
  Deny from all
</FilesMatch>

# Block wp-config backups
<FilesMatch "wp-config\.(bak|orig|save|swp|tmp)">
  Order allow,deny
  Deny from all
</FilesMatch>

# ----- 6. Block PHP execution in uploads -----
<Directory "/www/uletnayaparkovka.ru/wp-content/uploads">
  <FilesMatch "\.php$">
    Order allow,deny
    Deny from all
  </FilesMatch>
</Directory>

# ----- 7. Запрет XML-RPC (если не используется) -----
<Files xmlrpc.php>
  Order allow,deny
  Deny from all
</Files>
